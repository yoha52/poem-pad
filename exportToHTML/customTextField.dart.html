import 'package:flutter/material.dart';
import 'package:flutter/rendering.dart';

class customTextField extends StatefulWidget {
  const customTextField({super.key});

  @override
  State<customTextField> createState() => _customTextFieldState();
}

class _customTextFieldState extends State<customTextField> {
  final TextEditingController _textController = TextEditingController();
  final ScrollController _scrollController1 = ScrollController();
  final ScrollController _scrollController2 = ScrollController();
  List <String> _lineletterslist=[];
  @override
  void initState() {
    super.initState();
    _updateLineCounter();
    _textController.addListener(_updateLineCounter);
    _scrollController1.addListener(controller1Listener);
    _scrollController2.addListener(controller2Listener);
  }
  void controller1Listener() {
    if (_scrollController1.offset != _scrollController2.offset) {
      setState(() {
        _scrollController2.jumpTo(_scrollController1.offset);
      });
    }
  }
  void controller2Listener() {
    if (_scrollController1.offset != _scrollController2.offset) {
      setState(() {
        _scrollController1.jumpTo(_scrollController2.offset);
      });
    }
  }

  @override
  void dispose() {
    _textController.removeListener(_updateLineCounter);
    _textController.dispose();
    _scrollController1.dispose();
    _scrollController2.dispose();
    super.dispose();
  }

  void _updateLineCounter() {
    final lineletter = _textController.text.split('\n');
    if (lineletter != _lineletterslist) {
      setState(() {
        _lineletterslist = lineletter;
      });
    }
  }

  int _countLexicalVowels(String text) {
    int count = 0;
    if(text.isEmpty) return 0;
    final amharicRange = RegExp(r'[\u1200-\u137F]');
    if(amharicRange.hasMatch(text)){
      String cleanLine = text.replaceAll(RegExp(r'[\u1360-\u137F\s]+'), ' ').trim();
      if (cleanLine.isEmpty) return 0;

      // Split into individual words to safely apply position-based Sades rules
      List<String> words = cleanLine.split(' ');
      int totalSyllables = 0;

      // Define all 6th order (ሳድስ) characters in Unicode
      final sadesSet = {
        'ህ', 'ል', 'ሐ', 'ም', 'ሥ', 'ር', 'ስ', 'ሸ', 'ቀ', 'ብ', 'ት', 'ች', 'ን', 'ኝ','ዕ',
        'እ', 'ክ', 'ቭ', 'ዝ', 'ዥ', 'ይ', 'ድ', 'ጅ', 'ግ', 'ጥ', 'ጭ', 'ጵ', 'ጽ', 'ፅ', 'ፍ', 'ፕ'
      };

      for (var word in words) {
        if (word.isEmpty) continue;

        int wordSyllables = 0;

        for (int i = 0; i < word.length; i++) {
          String char = word[i];

          if (sadesSet.contains(char)) {
            // Rule 3: Sades at the very start of a word counts as 1
            if (i == 0) {
              wordSyllables++;
            }
            // Rule 2: Double sadeses (two consecutive sades characters) count as 1
            else if (sadesSet.contains(word[i - 1])) {
              // If the previous character was also a Sades, it combines into 1 syllable.
              // We already didn't count the previous inside-Sades (Rule 1), so we add 1 now.
              wordSyllables++;
            }
            // Rule 1: Sades inside another letter (middle or end of word) doesn't count
            else {
              // Do nothing (counts as 0)
            }
          } else {
            // Rule 4: Standalone vowels (አ, ኡ, ኢ, etc.) and all other standard 1st-7th orders count as 1
            wordSyllables++;
          }
        }

        totalSyllables += wordSyllables;
      }

      return totalSyllables;
    }

    return text.toLowerCase().replaceAll(RegExp(r'[^a-z\s]'), '').split(RegExp(r'\s+')).fold(0, (sum, w) {
      if (w.isEmpty) return sum;
      // 1. Base count using vowel clusters
      int c = RegExp(r'[aeiouy]+').allMatches(w).length;
      // 2. Adjust for trailing silent 'e' (but keep for consonant + 'le' like candle)
      if (w.endsWith('e') && !(w.endsWith('le') && w.length > 2 && !'aeiouy'.contains(w[w.length - 3]))) c--;
      // 3. Precise adjustments for common poetic words (di-et, po-et, rhi-no)
      if (RegExp(r'(?:[aeiou]ing|diet|poet|tri[au]|scien|dual|ide|flu|[^g]et)$').hasMatch(w)) c++;
      return sum + (c <= 0 ? 1 : c);
    });
  }
  @override
  Widget build(BuildContext context) {
    const fontsize = 16.0;
    const fontfamily = 'monospace';

    return Expanded(
      child: Padding(
        padding: const EdgeInsets.all(8.0),
        child: Container(
          decoration: BoxDecoration(
            borderRadius: BorderRadius.circular(20),
            color: Colors.white70,
          ),
          child: Padding(
            padding: const EdgeInsets.symmetric(vertical: 10,horizontal: 10),
            child: Row(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                Container(
                  width: 50,
                  padding: EdgeInsets.symmetric(vertical: 10),
                  decoration: BoxDecoration(
                    borderRadius: BorderRadius.circular(10),
                    color: Colors.grey.shade400,
                  ),
                  child: ScrollConfiguration(
                    behavior: ScrollConfiguration.of(context).copyWith(
                      scrollbars: false
                    ),
                    child: Padding(
                      padding: const EdgeInsets.only(right: 5),
                      child: ListView.builder(
                        controller: _scrollController1,
                        physics: const ClampingScrollPhysics(),
                        itemCount: _lineletterslist.length,
                        scrollCacheExtent: ScrollCacheExtent.viewport(0.5),
                        itemBuilder: (context, index)
                        {
                          dynamic x=_countLexicalVowels(_lineletterslist[index]);
                          if(x==0){
                            x='';
                          }
                          return Container(
                            alignment: Alignment.centerRight,
                            padding: EdgeInsets.only(right: 1),
                            height: (fontsize * 1.5),
                            child: Container(
                              child: Text(
                                '$x',
                                style: TextStyle(
                                  fontSize: fontsize,
                                  color: Colors.black,
                                  fontFamily: fontfamily,
                                ),
                              ),
                            ),
                          );
                        },
                      ),
                    ),
                  ),
                ),
                Expanded(
                  child: TextField(
                    controller: _textController,
                    scrollController: _scrollController2,
                    keyboardType: TextInputType.multiline,
                    scrollPhysics: const ClampingScrollPhysics(),
                    maxLines: null,
                    style: TextStyle(
                      fontSize: fontsize,
                      fontFamily: fontfamily,
                      color: Colors.black,
                    ),
                    decoration: const InputDecoration(
                        border: InputBorder.none,
                        contentPadding: EdgeInsets.symmetric(
                            vertical: 11,
                            horizontal: 5
                        )
                    ),
                  ),
                ),
              ],
            ),
          ),
        ),
      ),
    );
  }
